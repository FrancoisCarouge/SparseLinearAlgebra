43 // Not a constant expression: every operation below is `constexpr`, and this
44 // test would otherwise be evaluated at compile time, and never at run time.

49 using sparse::diagonal;
50
51 constexpr double infinity{std::numeric_limits<double>::infinity()};
52 constexpr double not_a_number{std::numeric_limits<double>::quiet_NaN()};
53
54 const sparse_matrix<double, diagonal<2>> value{1., 2.};
55 const sparse_matrix<double, diagonal<2>> unbounded{infinity, 1.};
56 const sparse_matrix<double, diagonal<2>> invalid{not_a_number, 1.};
57
58 // Scaling.
59 const auto scaled{value * infinity};
60 const auto dense_scaled{sparse_matrix<double, sparse::dense<2, 2>>{value} *
61 infinity};
62
63 assert((scaled.at<0, 0>() == infinity));
64 assert((scaled.at<0, 1>() == 0.));
65 assert((dense_scaled.at<0, 1>() != dense_scaled.at<0, 1>()));
66
67 // Dividing by zero.
68 const auto divided{value / 0.};
69
70 assert((divided.at<0, 0>() == infinity));
71 assert((divided.at<0, 1>() == 0.));
72
73 // Dividing by a matrix of blocks of a single unknown.
74 using dense_matrix = sparse_matrix<double, sparse::dense<2, 2>>;
75
76 const auto quotient{unbounded / value};
77 const auto dense_quotient{dense_matrix{unbounded} / dense_matrix{value}};
78
79 assert((quotient.at<0, 0>() == infinity));
80 assert((quotient.at<0, 1>() == 0.));
81 assert((quotient.at<1, 1>() == 0.5));
82 assert((dense_quotient.at<0, 1>() != dense_quotient.at<0, 1>()));
83
84 // A product.
85 const auto product{unbounded * value};
86
87 assert((product.at<0, 0>() == infinity));
88 assert((product.at<0, 1>() == 0.));
89 assert((product.at<1, 0>() == 0.));
90
91 // A product of more elementary products than a product expands: the terms of
92 // the structural zeros are not computed either.
93 using band_matrix = sparse_matrix<double, sparse::banded<48, 48, 8, 8>>;
94
95 band_matrix ones;
96
97 for (double &element : ones.values) {
98 element = 1.;
99 }

101 band_matrix unbounded_band{ones};
102
103 unbounded_band.at<0, 0>() = infinity;
104
105 const auto large_product{unbounded_band * ones};
106 const auto dense_large_product{

109
110 assert((large_product.at<0, 0>() == infinity));
111 assert((large_product.at<0, 16>() == 1.));
112 assert((large_product.at<0, 17>() == 0.));
113 assert((dense_large_product.at<0, 16>() != dense_large_product.at<0, 16>()));
114
115 // A stored not-a-number value stays visible.
