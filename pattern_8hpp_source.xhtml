130 [[nodiscard]] constexpr auto lower_bound(position value) const
131 -> std::size_t {
132 std::size_t first{0};
133 std::size_t size{Count};
134
135 while (size > 0) {
136 const std::size_t half{size / 2};
137
138 if (positions[first + half] < value) {
139 first += half + 1;
140 size -= half + 1;
141 } else {
142 size = half;
143 }
144 }
145
146 return first;
147 }
148
153 [[nodiscard]] constexpr auto index_of(std::size_t row,
154 std::size_t column) const
155 -> std::size_t {
156 const position value{row, column};
157 const std::size_t index{lower_bound(value)};
158
159 return index < Count && positions[index] == value ? index : absent;
160 }
161
165 [[nodiscard]] constexpr auto stores(std::size_t row, std::size_t column) const
166 -> bool {
167 return index_of(row, column) != absent;
168 }
169};

202template <std::size_t Rows, std::size_t Columns, std::size_t Count>
203[[nodiscard]] consteval auto make_pattern(const position (&positions)[Count])
204 -> pattern<Rows, Columns, Count> {
205 pattern<Rows, Columns, Count> result;
206
207 for (std::size_t i{0}; i < Count; ++i) {
208 const position value{positions[i]};
209 std::size_t j{i};
210
211 if (value.row >= Rows || value.column >= Columns) {
212 sparse_linear_algebra::internal::position_out_of_range_or_duplicate();
213 }
214
215 for (; j > 0 && value < result.positions[j - 1]; --j) {
216 result.positions[j] = result.positions[j - 1];
217 }
218
219 if (j > 0 && value == result.positions[j - 1]) {
220 sparse_linear_algebra::internal::position_out_of_range_or_duplicate();
221 }
222
223 result.positions[j] = value;
224 }
225
226 return result;
227}
