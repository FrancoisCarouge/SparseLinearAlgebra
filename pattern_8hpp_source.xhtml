68template <typename Predicate>
69concept position_predicate =
70 std::is_empty_v<Predicate> and std::default_initializable<Predicate> and
71 std::predicate<const Predicate &, std::size_t, std::size_t>;

80template <std::size_t Rows, std::size_t Columns, position_predicate Predicate>
81[[nodiscard]] consteval auto count_of() -> std::size_t {
82 const Predicate predicate{};
83 std::size_t result{0};
84
85 for (std::size_t row{0}; row < Rows; ++row) {
86 for (std::size_t column{0}; column < Columns; ++column) {
87 result += predicate(row, column) ? std::size_t{1} : std::size_t{0};
88 }
89 }
90
91 return result;
92}

155template <std::size_t Rows, std::size_t Columns, std::size_t Count>
156struct pattern {
158 std::size_t rows{Rows};
159
161 std::size_t columns{Columns};
162
164 std::size_t count{Count};
165
170 std::array<position, (Count > 0 ? Count : 1)> positions{};
171
175 consteval explicit pattern([[maybe_unused]] shape_t<Rows, Columns> dimensions)
176 requires(Count == 0)
177 {}
178
192 template <std::size_t Size>
193 requires(Size == Count)
194 consteval pattern([[maybe_unused]] shape_t<Rows, Columns> dimensions,
195 const position (&stored)[Size]) {
196 for (std::size_t i{0}; i < Count; ++i) {
197 const position value{stored[i]};
198 std::size_t j{i};
199
200 if (value.row >= Rows || value.column >= Columns) {
201 sparse_linear_algebra::internal::position_out_of_range_or_duplicate();
202 }
203
204 for (; j > 0 && value < positions[j - 1]; --j) {
205 positions[j] = positions[j - 1];
206 }
207
208 if (j > 0 && value == positions[j - 1]) {
209 sparse_linear_algebra::internal::position_out_of_range_or_duplicate();
210 }
211
212 positions[j] = value;
213 }
214 }
215
222 template <sparse_linear_algebra::internal::position_predicate Predicate>
223 requires(
224 sparse_linear_algebra::internal::count_of<Rows, Columns, Predicate>() ==
225 Count)
226 consteval pattern([[maybe_unused]] shape_t<Rows, Columns> dimensions,
227 const Predicate &predicate) {
228 std::size_t index{0};
229
230 for (std::size_t row{0}; row < Rows; ++row) {
231 for (std::size_t column{0}; column < Columns; ++column) {
232 if (predicate(row, column)) {
233 positions[index] = {row, column};
234 ++index;
235 }
236 }
237 }
238 }

243 [[nodiscard]] constexpr auto lower_bound(position value) const
244 -> std::size_t {
245 std::size_t first{0};
246 std::size_t size{Count};
247
248 while (size > 0) {
249 const std::size_t half{size / 2};
250
251 if (positions[first + half] < value) {
252 first += half + 1;
253 size -= half + 1;
254 } else {
255 size = half;
256 }
257 }
258
259 return first;
260 }
261
266 [[nodiscard]] constexpr auto index_of(std::size_t row,
267 std::size_t column) const
268 -> std::size_t {
269 const position value{row, column};
270 const std::size_t index{lower_bound(value)};
271
272 return index < Count && positions[index] == value ? index : absent;
273 }
274
278 [[nodiscard]] constexpr auto stores(std::size_t row, std::size_t column) const
279 -> bool {
280 return index_of(row, column) != absent;
281 }
282};

286template <std::size_t LhsRows, std::size_t LhsColumns, std::size_t LhsCount,
