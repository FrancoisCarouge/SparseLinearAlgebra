113template <std::size_t Position, typename Matrix, typename Value>
114constexpr void write_given(Matrix &result, const Value &value) {
115 if constexpr (not std::same_as<Value, structural_zero>) {
116 static constexpr std::size_t index{
117 Matrix::structure.index_of(Position / Matrix::structure.columns,
118 Position % Matrix::structure.columns)};
119
120 // The implicit conversion the constraints verified, stated: a conversion
121 // to the element type may narrow, or promote, which braces reject or an
122 // implicit conversion warns of.
123 result.values[index] = static_cast<element_t<Matrix>>(value);
124 }
125}

147template <typename Matrix, typename... Values>
148 requires sla::is_matrix<Matrix> and
149 (sizeof...(Values) ==
150 Matrix::structure.rows * Matrix::structure.columns) and
151 sla::are_given<Matrix, Values...>
152[[nodiscard]] constexpr auto make_matrix(const Values &...values) -> Matrix {
153 Matrix result;
154
155 [&result, &values...]<std::size_t... Positions>(
156 std::index_sequence<Positions...>) -> void {
157 (sla::write_given<Positions>(result, values), ...);
158 }(std::index_sequence_for<Values...>{});
159
160 return result;
161}
