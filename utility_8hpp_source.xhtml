271template <auto Lhs, auto Rhs, bool KeepAll>
272[[nodiscard]] consteval auto merge_positions() {
273 position_list<Lhs.count + Rhs.count> list;
274 std::size_t i{0};
275 std::size_t j{0};
276
277 while (i < Lhs.count || j < Rhs.count) {
278 if (j == Rhs.count ||
279 (i < Lhs.count && Lhs.positions[i] < Rhs.positions[j])) {
280 if constexpr (KeepAll) {
281 list.push(Lhs.positions[i]);
282 }
283 ++i;
284 } else if (i == Lhs.count || Rhs.positions[j] < Lhs.positions[i]) {
285 if constexpr (KeepAll) {
286 list.push(Rhs.positions[j]);
287 }
288 ++j;
289 } else {
290 list.push(Lhs.positions[i]);
291 ++i;
292 ++j;
293 }
294 }
295
296 return list;
297}

370template <auto Lhs, auto Rhs> [[nodiscard]] consteval auto reached_columns() {
371 using row_set = std::array<std::uint64_t, word_count<Rhs>>;
372
373 std::array<row_set, row_count<Rhs>> selected{};
374
375 for (std::size_t i{0}; i < Rhs.count; ++i) {
376 const auto [row, column]{Rhs.positions[i]};
377
378 selected[row][column / 64] |= std::uint64_t{1} << (column % 64);
379 }
380
381 std::array<row_set, row_count<Lhs>> result{};
382
383 for (std::size_t i{0}; i < Lhs.count; ++i) {
384 const auto [row, inner]{Lhs.positions[i]};
385
386 for (std::size_t word{0}; word < word_count<Rhs>; ++word) {
387 result[row][word] |= selected[inner][word];
388 }
389 }
390
391 return result;
392}

411 position_list<product_position_count<Lhs, Rhs>> list;

614template <auto Pattern>
615[[nodiscard]] consteval auto augment(std::size_t start,
616 matching<Pattern.columns> &matched)
617 -> bool {
618 std::array<bool, Pattern.columns> visited{};
619 std::array<std::size_t, Pattern.columns> previous{};
620 std::array<std::size_t, Pattern.columns> queue{};
621 std::size_t head{0};
622 std::size_t tail{0};
623 std::size_t row{start};
624 std::size_t from{sparse::absent};
625
626 while (row != sparse::absent) {
627 const std::size_t end{Pattern.lower_bound({row + 1, 0})};
628
629 for (std::size_t i{Pattern.lower_bound({row, 0})}; i < end; ++i) {
630 const std::size_t column{Pattern.positions[i].column};
631
632 if (not visited[column]) {
633 visited[column] = true;
634 previous[column] = from;
635
636 if (matched.row[column] == sparse::absent) {
637 for (std::size_t current{column}; current != sparse::absent;
638 current = previous[current]) {
639 matched.row[current] = previous[current] == sparse::absent
640 ? start
641 : matched.row[previous[current]];
642 }
643
644 return true;
645 }
646
647 queue[tail] = column;
648 ++tail;
649 }
650 }
651
652 row = sparse::absent;
653
654 if (head < tail) {
655 from = queue[head];
656 row = matched.row[from];
657 ++head;
658 }
659 }
660
661 return false;
662}

669template <auto Pattern>
670[[nodiscard]] consteval auto maximum_matching() -> matching<Pattern.columns> {
671 matching<Pattern.columns> result{};
672 std::array<bool, Pattern.rows> placed{};
673
674 for (std::size_t column{0}; column < Pattern.columns; ++column) {
675 result.row[column] = sparse::absent;
676 }
677
678 for (std::size_t row{0}; row < Pattern.rows; ++row) {
679 if (row < Pattern.columns && Pattern.stores(row, row)) {
680 result.row[row] = row;
681 placed[row] = true;
682 ++result.size;
683 }
684 }
685
686 for (std::size_t row{0}; row < Pattern.rows; ++row) {
687 if (not placed[row] && augment<Pattern>(row, result)) {
688 ++result.size;
689 }
690 }
691
692 return result;
693}

738template <std::size_t Size>
739consteval void enter_component(std::size_t unknown,
740 component_search<Size> &search) {
741 ++search.visits;
742 search.index[unknown] = search.visits;
743 search.lowest[unknown] = search.visits;
744 search.stack[search.stack_size] = unknown;
745 ++search.stack_size;
746 search.stacked[unknown] = true;
747}

751template <std::size_t Size>
752consteval void close_component(std::size_t unknown,
753 component_search<Size> &search,
754 block_triangular<Size> &result) {
755 const std::size_t first{search.ordered};
756 std::size_t member{sparse::absent};
757
758 result.begin[result.blocks] = first;
759
760 while (member != unknown) {
761 --search.stack_size;
762 member = search.stack[search.stack_size];
763 search.stacked[member] = false;
764 result.order[search.ordered] = member;
765 result.block[member] = result.blocks;
766 ++search.ordered;
767 }
768
769 // The unknowns of a block in ascending order: the elimination of a block
770 // follows the order of the system.
771 for (std::size_t i{first + 1}; i < search.ordered; ++i) {
772 const std::size_t moved{result.order[i]};
773 std::size_t j{i};
774
775 while (j > first && result.order[j - 1] > moved) {
776 result.order[j] = result.order[j - 1];
777 --j;
778 }
779
780 result.order[j] = moved;
781 }
782
783 for (std::size_t i{first}; i < search.ordered; ++i) {
784 result.rank[result.order[i]] = i;
785 }
786
787 ++result.blocks;
788 result.begin[result.blocks] = search.ordered;
789}

797template <auto Transposed>
798consteval void visit_component(std::size_t root,
799 component_search<Transposed.rows> &search,
800 block_triangular<Transposed.rows> &result) {
801 std::array<std::size_t, Transposed.rows> visiting{};
802 std::array<std::size_t, Transposed.rows> next{};
803 std::size_t depth{1};
804
805 enter_component(root, search);
806 visiting[0] = root;
807 next[0] = Transposed.lower_bound({result.equation[root], 0});
808
809 while (depth > 0) {
810 const std::size_t unknown{visiting[depth - 1]};
811 const std::size_t end{
812 Transposed.lower_bound({result.equation[unknown] + 1, 0})};
813
814 if (next[depth - 1] < end) {
815 const std::size_t dependency{
816 Transposed.positions[next[depth - 1]].column};
817
818 ++next[depth - 1];
819
820 if (search.index[dependency] == 0) {
821 enter_component(dependency, search);
822 visiting[depth] = dependency;
823 next[depth] = Transposed.lower_bound({result.equation[dependency], 0});
824 ++depth;
825 } else if (search.stacked[dependency] &&
826 search.index[dependency] < search.lowest[unknown]) {
827 search.lowest[unknown] = search.index[dependency];
828 }
829 } else {
830 if (search.lowest[unknown] == search.index[unknown]) {
831 close_component(unknown, search, result);
832 }
833
834 --depth;
835
836 if (depth > 0 &&
837 search.lowest[unknown] < search.lowest[visiting[depth - 1]]) {
838 search.lowest[visiting[depth - 1]] = search.lowest[unknown];
839 }
840 }
841 }
842}

846template <auto Pattern>
847inline constexpr auto block_triangular_form{[] consteval {
848 constexpr auto transposed{to_pattern<Pattern.columns, Pattern.rows,
849 transpose_positions<Pattern>()>()};
850 constexpr auto matched{maximum_matching<Pattern>()};
851 block_triangular<Pattern.rows> result{};
852 component_search<Pattern.rows> search{};
853
854 for (std::size_t column{0}; column < Pattern.columns; ++column) {
855 result.equation[matched.row[column]] = column;
856 }
857
858 for (std::size_t unknown{0}; unknown < Pattern.rows; ++unknown) {
859 if (search.index[unknown] == 0) {
860 visit_component<transposed>(unknown, search, result);
861 }
862 }
863
864 return result;
865}()};

873template <auto Lhs, auto Rhs>
874inline constexpr auto quotient_reach{[] {
875 constexpr auto form{block_triangular_form<Rhs>};
876 constexpr auto transposed{
877 to_pattern<Rhs.columns, Rhs.rows, transpose_positions<Rhs>()>()};
878 constexpr auto lhs_transposed{
879 to_pattern<Lhs.columns, Lhs.rows, transpose_positions<Lhs>()>()};
880 std::array<std::array<bool, Lhs.rows>, Rhs.rows> result{};
881
882 for (std::size_t block{0}; block < form.blocks; ++block) {

939template <auto Pattern> [[nodiscard]] consteval auto cholesky_columns() {
940 using row_set = std::array<std::uint64_t, word_count<Pattern>>;
941
942 std::array<row_set, row_count<Pattern>> result{};
943 std::array<std::size_t, row_count<Pattern>> parent{};
944 std::array<std::size_t, row_count<Pattern>> reached_by{};
945
946 for (std::size_t row{0}; row < Pattern.rows; ++row) {
947 const std::size_t end{Pattern.lower_bound({row + 1, 0})};
948
949 parent[row] = sparse::absent;
950 reached_by[row] = row;
951
952 for (std::size_t i{Pattern.lower_bound({row, 0})};
953 i < end && Pattern.positions[i].column < row; ++i) {
954 for (std::size_t column{Pattern.positions[i].column};
955 reached_by[column] != row; column = parent[column]) {
956 if (parent[column] == sparse::absent) {
957 parent[column] = row;
958 }
959
960 reached_by[column] = row;
961 result[row][column / 64] |= std::uint64_t{1} << (column % 64);
962 }
963 }
964
965 result[row][row / 64] |= std::uint64_t{1} << (row % 64);
966 }
967
968 return result;
969}

1013template <std::floating_point Type>
1014[[nodiscard]] constexpr auto square_root(Type value) -> Type {
1015#if __STDC_HOSTED__
1016 if (not std::is_constant_evaluated()) {
1017 return std::sqrt(value);
1018 }
1019#endif
1020
1021 // A zero, a negative, a not-a-number, or an infinite value has no iteration.
1022 if (not(value > Type{0}) || value > std::numeric_limits<Type>::max()) {
1023 return value;
1024 }
1025
1026 Type estimate{value < Type{1} ? Type{1} : value};
1027 Type next{(estimate + (value / estimate)) / Type{2}};
1028
1029 // The estimates decrease strictly until the root, where the iteration stops.
1030 while (next < estimate) {
1031 estimate = next;
1032 next = (estimate + (value / estimate)) / Type{2};
1033 }
1034
1035 return estimate;
1036}
