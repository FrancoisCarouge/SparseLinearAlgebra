262template <auto Lhs, auto Rhs, bool KeepAll>
263[[nodiscard]] consteval auto merge_positions() {
264 position_list<Lhs.count + Rhs.count> list;
265 std::size_t i{0};
266 std::size_t j{0};
267
268 while (i < Lhs.count || j < Rhs.count) {
269 if (j == Rhs.count ||
270 (i < Lhs.count && Lhs.positions[i] < Rhs.positions[j])) {
271 if constexpr (KeepAll) {
272 list.push(Lhs.positions[i]);
273 }
274 ++i;
275 } else if (i == Lhs.count || Rhs.positions[j] < Lhs.positions[i]) {
276 if constexpr (KeepAll) {
277 list.push(Rhs.positions[j]);
278 }
279 ++j;
280 } else {
281 list.push(Lhs.positions[i]);
282 ++i;
283 ++j;
284 }
285 }
286
287 return list;
288}

323template <auto Lhs, auto Rhs> [[nodiscard]] consteval auto product_positions() {
324 position_list<term_count<Lhs, Rhs>> list;
325
326 for (std::size_t row{0}; row < Lhs.rows; ++row) {
327 std::array<bool, Rhs.columns> reached{};
328 const std::size_t end{Lhs.lower_bound({row + 1, 0})};
329
330 for (std::size_t i{Lhs.lower_bound({row, 0})}; i < end; ++i) {
331 const std::size_t inner{Lhs.positions[i].column};
332 const std::size_t inner_end{Rhs.lower_bound({inner + 1, 0})};
333
334 for (std::size_t j{Rhs.lower_bound({inner, 0})}; j < inner_end; ++j) {
335 reached[Rhs.positions[j].column] = true;
336 }
337 }
338
339 for (std::size_t column{0}; column < Rhs.columns; ++column) {
340 if (reached[column]) {
341 list.push({row, column});
342 }
343 }
344 }
345
346 return list;
347}

538template <auto Pattern>
539[[nodiscard]] consteval auto augment(std::size_t start,
540 matching<Pattern.columns> &matched)
541 -> bool {
542 std::array<bool, Pattern.columns> visited{};
543 std::array<std::size_t, Pattern.columns> previous{};
544 std::array<std::size_t, Pattern.columns> queue{};
545 std::size_t head{0};
546 std::size_t tail{0};
547 std::size_t row{start};
548 std::size_t from{sparse::absent};
549
550 while (row != sparse::absent) {
551 const std::size_t end{Pattern.lower_bound({row + 1, 0})};
552
553 for (std::size_t i{Pattern.lower_bound({row, 0})}; i < end; ++i) {
554 const std::size_t column{Pattern.positions[i].column};
555
556 if (not visited[column]) {
557 visited[column] = true;
558 previous[column] = from;
559
560 if (matched.row[column] == sparse::absent) {
561 for (std::size_t current{column}; current != sparse::absent;
562 current = previous[current]) {
563 matched.row[current] = previous[current] == sparse::absent
564 ? start
565 : matched.row[previous[current]];
566 }
567
568 return true;
569 }
570
571 queue[tail] = column;
572 ++tail;
573 }
574 }
575
576 row = sparse::absent;
577
578 if (head < tail) {
579 from = queue[head];
580 row = matched.row[from];
581 ++head;
582 }
583 }
584
585 return false;
586}

593template <auto Pattern>
594[[nodiscard]] consteval auto maximum_matching() -> matching<Pattern.columns> {
595 matching<Pattern.columns> result{};
596 std::array<bool, Pattern.rows> placed{};
597
598 for (std::size_t column{0}; column < Pattern.columns; ++column) {
599 result.row[column] = sparse::absent;
600 }
601
602 for (std::size_t row{0}; row < Pattern.rows; ++row) {
603 if (row < Pattern.columns && Pattern.stores(row, row)) {
604 result.row[row] = row;
605 placed[row] = true;
606 ++result.size;
607 }
608 }
609
610 for (std::size_t row{0}; row < Pattern.rows; ++row) {
611 if (not placed[row] && augment<Pattern>(row, result)) {
612 ++result.size;
613 }
614 }
615
616 return result;
617}

662template <std::size_t Size>
663consteval void enter_component(std::size_t unknown,
664 component_search<Size> &search) {
665 ++search.visits;
666 search.index[unknown] = search.visits;
667 search.lowest[unknown] = search.visits;
668 search.stack[search.stack_size] = unknown;
669 ++search.stack_size;
670 search.stacked[unknown] = true;
671}

675template <std::size_t Size>
676consteval void close_component(std::size_t unknown,
677 component_search<Size> &search,
678 block_triangular<Size> &result) {
679 const std::size_t first{search.ordered};
680 std::size_t member{sparse::absent};
681
682 result.begin[result.blocks] = first;
683
684 while (member != unknown) {
685 --search.stack_size;
686 member = search.stack[search.stack_size];
687 search.stacked[member] = false;
688 result.order[search.ordered] = member;
689 result.block[member] = result.blocks;
690 ++search.ordered;
691 }
692
693 // The unknowns of a block in ascending order: the elimination of a block
694 // follows the order of the system.
695 for (std::size_t i{first + 1}; i < search.ordered; ++i) {
696 const std::size_t moved{result.order[i]};
697 std::size_t j{i};
698
699 while (j > first && result.order[j - 1] > moved) {
700 result.order[j] = result.order[j - 1];
701 --j;
702 }
703
704 result.order[j] = moved;
705 }
706
707 for (std::size_t i{first}; i < search.ordered; ++i) {
708 result.rank[result.order[i]] = i;
709 }
710
711 ++result.blocks;
712 result.begin[result.blocks] = search.ordered;
713}

721template <auto Transposed>
722consteval void visit_component(std::size_t root,
723 component_search<Transposed.rows> &search,
724 block_triangular<Transposed.rows> &result) {
725 std::array<std::size_t, Transposed.rows> visiting{};
726 std::array<std::size_t, Transposed.rows> next{};
727 std::size_t depth{1};
728
729 enter_component(root, search);
730 visiting[0] = root;
731 next[0] = Transposed.lower_bound({result.equation[root], 0});
732
733 while (depth > 0) {
734 const std::size_t unknown{visiting[depth - 1]};
735 const std::size_t end{
736 Transposed.lower_bound({result.equation[unknown] + 1, 0})};
737
738 if (next[depth - 1] < end) {
739 const std::size_t dependency{
740 Transposed.positions[next[depth - 1]].column};
741
742 ++next[depth - 1];
743
744 if (search.index[dependency] == 0) {
745 enter_component(dependency, search);
746 visiting[depth] = dependency;
747 next[depth] = Transposed.lower_bound({result.equation[dependency], 0});
748 ++depth;
749 } else if (search.stacked[dependency] &&
750 search.index[dependency] < search.lowest[unknown]) {
751 search.lowest[unknown] = search.index[dependency];
752 }
753 } else {
754 if (search.lowest[unknown] == search.index[unknown]) {
755 close_component(unknown, search, result);
756 }
757
758 --depth;
759
760 if (depth > 0 &&
761 search.lowest[unknown] < search.lowest[visiting[depth - 1]]) {
762 search.lowest[visiting[depth - 1]] = search.lowest[unknown];
763 }
764 }
765 }
766}

770template <auto Pattern>
771inline constexpr auto block_triangular_form{[] consteval {
772 constexpr auto transposed{to_pattern<Pattern.columns, Pattern.rows,
773 transpose_positions<Pattern>()>()};
774 constexpr auto matched{maximum_matching<Pattern>()};
775 block_triangular<Pattern.rows> result{};
776 component_search<Pattern.rows> search{};
777
778 for (std::size_t column{0}; column < Pattern.columns; ++column) {
779 result.equation[matched.row[column]] = column;
780 }
781
782 for (std::size_t unknown{0}; unknown < Pattern.rows; ++unknown) {
783 if (search.index[unknown] == 0) {
784 visit_component<transposed>(unknown, search, result);
785 }
786 }
787
788 return result;
789}()};

797template <auto Lhs, auto Rhs>
798inline constexpr auto quotient_reach{[] {
799 constexpr auto form{block_triangular_form<Rhs>};
800 constexpr auto transposed{
801 to_pattern<Rhs.columns, Rhs.rows, transpose_positions<Rhs>()>()};
802 constexpr auto lhs_transposed{
803 to_pattern<Lhs.columns, Lhs.rows, transpose_positions<Lhs>()>()};
804 std::array<std::array<bool, Lhs.rows>, Rhs.rows> result{};
805
806 for (std::size_t block{0}; block < form.blocks; ++block) {
807 std::array<bool, Lhs.rows> reached{};
808
809 for (std::size_t k{form.begin[block]}; k < form.begin[block + 1]; ++k) {
810 const std::size_t equation{form.equation[form.order[k]]};
811 const std::size_t lhs_end{lhs_transposed.lower_bound({equation + 1, 0})};
812 const std::size_t end{transposed.lower_bound({equation + 1, 0})};
813
814 for (std::size_t i{lhs_transposed.lower_bound({equation, 0})};
815 i < lhs_end; ++i) {
816 reached[lhs_transposed.positions[i].column] = true;
817 }
818
819 for (std::size_t i{transposed.lower_bound({equation, 0})}; i < end; ++i) {
820 const std::size_t dependency{transposed.positions[i].column};
821
822 for (std::size_t row{0}; row < Lhs.rows; ++row) {
823 reached[row] = reached[row] || result[dependency][row];
824 }
825 }
826 }
827
828 for (std::size_t k{form.begin[block]}; k < form.begin[block + 1]; ++k) {
829 result[form.order[k]] = reached;
830 }
831 }
832
833 return result;
834}()};

838template <auto Lhs, auto Rhs>
839[[nodiscard]] consteval auto quotient_positions() {

866[[nodiscard]] constexpr auto square_root(Type value) -> Type {
867#if __STDC_HOSTED__
868 if (not std::is_constant_evaluated()) {
869 return std::sqrt(value);
870 }
871#endif
872
873 // A zero, a negative, a not-a-number, or an infinite value has no iteration.
874 if (not(value > Type{0}) || value > std::numeric_limits<Type>::max()) {
875 return value;
876 }
877
878 Type estimate{value < Type{1} ? Type{1} : value};
879 Type next{(estimate + (value / estimate)) / Type{2}};
880
881 // The estimates decrease strictly until the root, where the iteration stops.
882 while (next < estimate) {
883 estimate = next;
884 next = (estimate + (value / estimate)) / Type{2};
885 }
886
887 return estimate;
888}
