48 template <auto Pattern>
49 [[nodiscard]] static constexpr auto fill(double seed)
50 -> sparse_matrix<double, Pattern> {
51 sparse_matrix<double, Pattern> result;
52
53 for (std::size_t i{0}; i < Pattern.count; ++i) {
54 const double value{seed + static_cast<double>(i)};
55
56 result.values.at(i) = i % 2 == 0 ? value : -value;
57 }
58
59 return result;
60 }

63 template <auto Pattern>
64 [[nodiscard]] static constexpr auto
65 to_reference(const sparse_matrix<double, Pattern> &value)
66 -> reference<Pattern.rows, Pattern.columns> {
67 reference<Pattern.rows, Pattern.columns> result{};
68
69 for (std::size_t row{0}; row < Pattern.rows; ++row) {
70 for (std::size_t column{0}; column < Pattern.columns; ++column) {
71 result.at(row).at(column) = value[row, column];
72 }
73 }
74
75 return result;
76 }

79 template <auto Lhs, auto Rhs>
80 requires(Lhs.rows == Rhs.rows and Lhs.columns == Rhs.columns)
81 static constexpr void verify_additive() {
82 const auto lhs{fill<Lhs>(3.)};
83 const auto rhs{fill<Rhs>(11.)};
84 const auto lhs_values{to_reference(lhs)};
85 const auto rhs_values{to_reference(rhs)};
86
87 const auto sum{to_reference(lhs + rhs)};
88 const auto difference{to_reference(lhs - rhs)};
89 const auto negated{to_reference(-lhs)};
90 const auto mirrored{to_reference(transposed(lhs))};
91
92 for (std::size_t row{0}; row < Lhs.rows; ++row) {
93 for (std::size_t column{0}; column < Lhs.columns; ++column) {
94 [[maybe_unused]] const double left{lhs_values.at(row).at(column)};
95 [[maybe_unused]] const double right{rhs_values.at(row).at(column)};
96
97 assert(sum.at(row).at(column) == left + right);
98 assert(difference.at(row).at(column) == left - right);
99 assert(negated.at(row).at(column) == -left);
100 assert(mirrored.at(column).at(row) == left);
101 }
102 }
103 }

106 template <auto Lhs, auto Rhs>
107 requires(Lhs.columns == Rhs.rows)
108 static constexpr void verify_product() {
109 const auto lhs{fill<Lhs>(3.)};
110 const auto rhs{fill<Rhs>(11.)};
111 const auto lhs_values{to_reference(lhs)};
112 const auto rhs_values{to_reference(rhs)};
113 const auto product{lhs * rhs};
114 const auto values{to_reference(product)};
115
116 for (std::size_t row{0}; row < Lhs.rows; ++row) {
117 for (std::size_t column{0}; column < Rhs.columns; ++column) {
118 double expected{0.};
119 bool reached{false};
120
121 for (std::size_t inner{0}; inner < Lhs.columns; ++inner) {
122 expected +=
123 lhs_values.at(row).at(inner) * rhs_values.at(inner).at(column);
124 reached =
125 reached || (Lhs.stores(row, inner) && Rhs.stores(inner, column));
126 }
127
128 assert(values.at(row).at(column) == expected);
129 assert(decltype(product)::structure.stores(row, column) == reached);
130 }
131 }
132 }

135 template <auto Lhs, auto Rhs> static constexpr void verify_dense() {
136 using dense_matrix =
137 sparse_matrix<double, sparse::dense<Lhs.rows, Lhs.columns>>;
138
139 const auto lhs{fill<Lhs>(3.)};
140 const auto rhs{fill<Rhs>(11.)};
141
142 assert((lhs + rhs) == (dense_matrix{lhs} + rhs));
143 assert((lhs - rhs) == (dense_matrix{lhs} - rhs));
144 assert((lhs * rhs) == (dense_matrix{lhs} * rhs));
145 assert(transposed(lhs) == transposed(dense_matrix{lhs}));
146 assert((lhs * 2.) == (dense_matrix{lhs} * 2.));
147 }

149 template <auto Lhs, auto... Patterns> static constexpr void verify_with() {
150 (verify_additive<Lhs, Patterns>(), ...);
151 (verify_product<Lhs, Patterns>(), ...);
152 (verify_dense<Lhs, Patterns>(), ...);
153 }

155 template <auto... Patterns> static constexpr void verify_all_pairs() {
156 (verify_with<Patterns, Patterns...>(), ...);
