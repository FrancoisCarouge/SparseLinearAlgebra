50 template <auto Pattern>
51 [[nodiscard]] static constexpr auto fill(double seed)
52 -> sparse_matrix<double, Pattern> {
53 sparse_matrix<double, Pattern> result;
54
55 for (std::size_t i{0}; i < Pattern.count; ++i) {
56 const double value{seed + static_cast<double>(i)};
57
58 result.values.at(i) = i % 2 == 0 ? value : -value;
59 }
60
61 return result;
62 }

65 template <auto Pattern>
66 [[nodiscard]] static constexpr auto
67 to_reference(const sparse_matrix<double, Pattern> &value)
68 -> reference<Pattern.rows, Pattern.columns> {
69 reference<Pattern.rows, Pattern.columns> result{};
70
71 for (std::size_t row{0}; row < Pattern.rows; ++row) {
72 for (std::size_t column{0}; column < Pattern.columns; ++column) {
73 result.at(row).at(column) = value[row, column];
74 }
75 }
76
77 return result;
78 }

81 template <auto Lhs, auto Rhs>
82 requires(Lhs.rows == Rhs.rows and Lhs.columns == Rhs.columns)
83 static constexpr void verify_additive() {
84 const auto lhs{fill<Lhs>(3.)};
85 const auto rhs{fill<Rhs>(11.)};
86 const auto lhs_values{to_reference(lhs)};
87 const auto rhs_values{to_reference(rhs)};
88
89 const auto sum{to_reference(lhs + rhs)};
90 const auto difference{to_reference(lhs - rhs)};
91 const auto negated{to_reference(-lhs)};
92 const auto mirrored{to_reference(transposed(lhs))};
93
94 for (std::size_t row{0}; row < Lhs.rows; ++row) {
95 for (std::size_t column{0}; column < Lhs.columns; ++column) {
96 [[maybe_unused]] const double left{lhs_values.at(row).at(column)};
97 [[maybe_unused]] const double right{rhs_values.at(row).at(column)};
98
99 assert(sum.at(row).at(column) == left + right);
100 assert(difference.at(row).at(column) == left - right);
101 assert(negated.at(row).at(column) == -left);
102 assert(mirrored.at(column).at(row) == left);
103 }
104 }
105 }

108 template <auto Lhs, auto Rhs>
109 requires(Lhs.columns == Rhs.rows)
110 static constexpr void verify_product() {
111 const auto lhs{fill<Lhs>(3.)};
112 const auto rhs{fill<Rhs>(11.)};
113 const auto lhs_values{to_reference(lhs)};
114 const auto rhs_values{to_reference(rhs)};
115 const auto product{lhs * rhs};
116 const auto values{to_reference(product)};
117
118 for (std::size_t row{0}; row < Lhs.rows; ++row) {
119 for (std::size_t column{0}; column < Rhs.columns; ++column) {
120 double expected{0.};
121 bool reached{false};
122
123 for (std::size_t inner{0}; inner < Lhs.columns; ++inner) {
124 expected +=
125 lhs_values.at(row).at(inner) * rhs_values.at(inner).at(column);
126 reached =
127 reached || (Lhs.stores(row, inner) && Rhs.stores(inner, column));
128 }
129
130 assert(values.at(row).at(column) == expected);
131 assert(decltype(product)::structure.stores(row, column) == reached);
132 }
133 }
134 }

138 template <auto Pattern>
139 [[nodiscard]] static constexpr auto fill_invertible()
140 -> sparse_matrix<double, Pattern> {
141 sparse_matrix<double, Pattern> result{fill<Pattern>(11.)};
142
143 for (std::size_t i{0}; i < Pattern.count; ++i) {
144 if (Pattern.positions.at(i).row == Pattern.positions.at(i).column) {
145 result.values.at(i) += 1000.;

154 template <std::size_t Rows, std::size_t Size>
155 [[nodiscard]] static auto solve(const reference<Rows, Size> &lhs,
156 const reference<Size, Size> &rhs)
157 -> reference<Rows, Size> {
158 reference<Size, Size> system{};
159 reference<Size, Rows> solution{};
160
161 for (std::size_t row{0}; row < Size; ++row) {
162 for (std::size_t column{0}; column < Size; ++column) {
163 system.at(row).at(column) = rhs.at(column).at(row);
164 }
165
166 for (std::size_t column{0}; column < Rows; ++column) {
167 solution.at(row).at(column) = lhs.at(column).at(row);
168 }
169 }
170
171 for (std::size_t pivot{0}; pivot < Size; ++pivot) {
172 std::size_t largest{pivot};
173
174 for (std::size_t row{pivot + 1}; row < Size; ++row) {
175 if (std::abs(system.at(row).at(pivot)) >
176 std::abs(system.at(largest).at(pivot))) {
177 largest = row;
178 }
179 }
180
181 std::swap(system.at(pivot), system.at(largest));
182 std::swap(solution.at(pivot), solution.at(largest));
183
184 for (std::size_t row{0}; row < Size; ++row) {
185 if (row != pivot) {
186 const double factor{system.at(row).at(pivot) /
187 system.at(pivot).at(pivot)};
188
189 for (std::size_t column{0}; column < Size; ++column) {
190 system.at(row).at(column) -= factor * system.at(pivot).at(column);
191 }
192
193 for (std::size_t column{0}; column < Rows; ++column) {
194 solution.at(row).at(column) -=
195 factor * solution.at(pivot).at(column);
196 }
197 }
198 }
199 }
200
201 reference<Rows, Size> result{};
202
203 for (std::size_t row{0}; row < Size; ++row) {
204 for (std::size_t column{0}; column < Rows; ++column) {
205 result.at(column).at(row) =
206 solution.at(row).at(column) / system.at(row).at(row);
207 }
208 }
209
210 return result;
211 }

215 template <auto Lhs, auto Rhs>
216 requires(Lhs.columns == Rhs.rows)
217 static void verify_quotient() {
218 const auto lhs{fill<Lhs>(3.)};
219 const auto rhs{fill_invertible<Rhs>()};
220 const auto quotient{lhs / rhs};
221 const auto values{to_reference(quotient)};
222 const auto expected{solve(to_reference(lhs), to_reference(rhs))};
223
224 for (std::size_t row{0}; row < Lhs.rows; ++row) {
225 for (std::size_t column{0}; column < Rhs.columns; ++column) {
226 [[maybe_unused]] const double value{values.at(row).at(column)};
227 [[maybe_unused]] const double reference_value{
228 expected.at(row).at(column)};
229
230 assert(std::abs(value - reference_value) <=
231 1e-12 * (1. + std::abs(reference_value)));
232 assert(decltype(quotient)::structure.stores(row, column) ||
233 std::abs(reference_value) <= 1e-12);
234 }
235 }
236 }

250 template <auto Lhs, auto Rhs> static constexpr void verify_dense() {
251 using dense_matrix =
252 sparse_matrix<double, sparse::dense<Lhs.rows, Lhs.columns>>;
253
254 const auto lhs{fill<Lhs>(3.)};
255 const auto rhs{fill<Rhs>(11.)};
256
257 assert((lhs + rhs) == (dense_matrix{lhs} + rhs));
258 assert((lhs - rhs) == (dense_matrix{lhs} - rhs));
259 assert((lhs * rhs) == (dense_matrix{lhs} * rhs));
260 assert(transposed(lhs) == transposed(dense_matrix{lhs}));
261 assert((lhs * 2.) == (dense_matrix{lhs} * 2.));
262 }
