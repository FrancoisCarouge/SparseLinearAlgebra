39struct counted {
40 double value{0.};
41 std::size_t multiplications{0};
42 std::size_t additions{0};

44 [[nodiscard]] friend constexpr auto operator*(const counted &lhs,
45 const counted &rhs) -> counted {
46 return {.value = lhs.value * rhs.value,
47 .multiplications = lhs.multiplications + rhs.multiplications + 1,
48 .additions = lhs.additions + rhs.additions};
49 }
50
51 [[nodiscard]] friend constexpr auto operator+(const counted &lhs,
52 const counted &rhs) -> counted {
53 return {.value = lhs.value + rhs.value,
54 .multiplications = lhs.multiplications + rhs.multiplications,
55 .additions = lhs.additions + rhs.additions + 1};
56 }
57
58 friend constexpr auto operator+=(counted &lhs, const counted &rhs)
59 -> counted & {
