39struct counted {
40 double value{0.};

42 std::size_t additions{0};

45 [[nodiscard]] friend constexpr auto operator*(const counted &lhs,
46 const counted &rhs) -> counted {
47 return {.value = lhs.value * rhs.value,
48 .multiplications = lhs.multiplications + rhs.multiplications + 1,
49 .additions = lhs.additions + rhs.additions,
50 .divisions = lhs.divisions + rhs.divisions};
51 }

104struct total {
105 template <auto Pattern>
106 [[nodiscard]] static consteval auto
107 multiplications(const storage<Pattern> &value) -> std::size_t {
108 std::size_t result{0};
109
110 for (const counted &element : value.values) {
111 result += element.multiplications;
112 }
113
114 return result;
115 }

143struct operand {
146 template <auto Pattern>
147 [[nodiscard]] static consteval auto distinct() -> storage<Pattern> {
148 storage<Pattern> result;
