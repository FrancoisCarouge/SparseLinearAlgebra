41 // Not a constant expression: every operation below is `constexpr`, and this
42 // test would otherwise be evaluated at compile time, and never at run time.
43 [[maybe_unused]] const volatile int run_time{0};
44
45 assert(run_time == 0);

47 using sparse::diagonal;
48 using sparse::make_matrix;
49 using sparse::make_pattern;
50 using sparse::literals::operator""_o;
51 using sparse::tridiagonal;
52
53 constexpr auto wide{make_pattern<2, 3>({{0, 0}, {0, 2}, {1, 1}})};
54 constexpr auto tall{make_pattern<3, 2>({{0, 1}, {1, 0}, {2, 1}})};
55 constexpr auto first_element{make_pattern<3, 1>({{0, 0}})};
56 constexpr auto row_pattern{make_pattern<1, 3>({{0, 0}, {0, 2}})};
57 constexpr auto column_pattern{make_pattern<3, 1>({{0, 0}, {1, 0}})};
58
59 const sparse_matrix<double, diagonal<3>> scale_matrix{2., 3., 4.};
60 const sparse_matrix<double, tridiagonal<3>> band{1., 2., 3., 4., 5., 6., 7.};
61
62 // Left scales the rows, right scales the columns: not commutative.
63 const auto left{scale_matrix * band};
64 const auto right{band * scale_matrix};
65
66 assert((left.at<0, 1>() == 4.));
67 assert((left.at<1, 0>() == 9.));
68 assert((left.at<2, 1>() == 24.));
69 assert((right.at<0, 1>() == 6.));
70 assert((right.at<1, 0>() == 6.));
71 assert((right.at<2, 1>() == 18.));
72 assert((not(left == right)));
73
74 // A band squared reaches further: every term of the middle row.
75 const auto square{band * band};
76
77 assert((square.at<0, 0>() == 7.));
78 assert((square.at<0, 1>() == 10.));
79 assert((square.at<0, 2>() == 10.));
80 assert((square.at<1, 1>() == 52.));
81 assert((square.at<2, 0>() == 18.));
82 assert((square.at<2, 2>() == 79.));
83
84 // A non-square product.
85 const auto lhs{
86 make_matrix<sparse_matrix<double, wide>>(1., 0_o, 2., 0_o, 3., 0_o)};
87 const auto rhs{
88 make_matrix<sparse_matrix<double, tall>>(0_o, 4., 5., 0_o, 0_o, 6.)};
89
90 const auto product{lhs * rhs};
91
92 static_assert(decltype(product)::structure.rows == 2 and
93 decltype(product)::structure.columns == 2);
94 assert((product.at<0, 0>() == 0.));
95 assert((product.at<0, 1>() == 16.));
96 assert((product.at<1, 0>() == 15.));
97 assert((product.at<1, 1>() == 0.));
98
99 const auto reversed{rhs * lhs};
100
101 static_assert(decltype(reversed)::structure.rows == 3 and
102 decltype(reversed)::structure.columns == 3);
103 assert((reversed.at<0, 1>() == 12.));
104 assert((reversed.at<1, 0>() == 5.));
105 assert((reversed.at<1, 2>() == 10.));
106 assert((reversed.at<2, 1>() == 18.));
107 assert((reversed.at<0, 0>() == 0.));
108
109 // A matrix by a vector: a sparse vector stays sparse.
110 const sparse_matrix<double, first_element> column{2.};
111 const auto image{band * column};
112
113 static_assert(decltype(image)::structure.count == 2);
114 assert((image.at<0, 0>() == 2.));
115 assert((image.at<1, 0>() == 6.));
116 assert((image.at<2, 0>() == 0.));
117
118 // A row by a column is a singleton, a column by a row is a matrix.
119 const sparse_matrix<double, row_pattern> row_vector{3., 4.};
120 const sparse_matrix<double, column_pattern> column_vector{5., 6.};
121
122 const auto inner{row_vector * column_vector};
123 const auto outer{column_vector * row_vector};
124
125 static_assert(decltype(inner)::structure.count == 1);
126 static_assert(decltype(outer)::structure.count == 4);
127 assert((inner.at<0, 0>() == 15.));
128 assert((outer.at<0, 0>() == 15.));
129 assert((outer.at<0, 2>() == 20.));
130 assert((outer.at<1, 2>() == 24.));
131 assert((outer.at<2, 2>() == 0.));
132
133 // Associative.
134 assert(((scale_matrix * band) * band == scale_matrix * (band * band)));
135 assert((transposed(scale_matrix * band) ==
136 transposed(band) * transposed(scale_matrix)));
137
138 // Distributive, over a wider pattern than either term.
139 assert((band * (scale_matrix + band) == band * scale_matrix + band * band));
140
141 // A zero matrix absorbs.
142 const auto none{band * sparse_matrix<double, sparse::zero<3, 3>>{}};
143
144 static_assert(decltype(none)::structure.count == 0);
145 assert((none.at<1, 1>() == 0.));
