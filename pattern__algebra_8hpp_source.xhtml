69template <sparsity_pattern auto Pattern>
70inline constexpr auto pattern_transpose{[] {
71 pattern<Pattern.columns, Pattern.rows, Pattern.count> result;
72 std::size_t index{0};
73
74 for (std::size_t column{0}; column < Pattern.columns; ++column) {
75 for (std::size_t i{0}; i < Pattern.count; ++i) {
76 if (Pattern.positions[i].column == column) {
77 result.positions[index] = {column, Pattern.positions[i].row};
78 ++index;
79 }
80 }
81 }
82
83 return result;
84}()};
