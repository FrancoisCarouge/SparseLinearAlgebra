46[[maybe_unused]] const auto test{[] -> int {
47 // Not a constant expression: every operation below is `constexpr`, and this
48 // test would otherwise be evaluated at compile time, and never at run time.
49 [[maybe_unused]] const volatile int run_time{0};
50
51 assert(run_time == 0);
52
53 // A singleton denominator divides each stored element.
54 const sparse_matrix<double, sparse::dense<2, 1>> column{{1.}, {-3.}};
55 const sparse_matrix<double, sparse::dense<1, 1>> singleton{{4.}};
56 const auto scaled{column / singleton};
57
58 assert((scaled.at<0, 0>() == 0.25));
59 assert((scaled.at<1, 0>() == -0.75));
60
61 // A diagonal denominator divides each column, and keeps the pattern.
62 using tridiagonal_matrix = sparse_matrix<double, sparse::tridiagonal<3>>;
63
64 const tridiagonal_matrix band{1., 2., 3., 4., 5., 6., 7.};
65 const sparse_matrix<double, sparse::diagonal<3>> diagonal{2., 4., 0.5};
66 const auto columns{band / diagonal};
67
68 static_assert(
69 std::same_as<std::remove_cvref_t<decltype(columns)>, tridiagonal_matrix>);
70 assert((columns.at<0, 0>() == 0.5));
71 assert((columns.at<0, 1>() == 0.5));
72 assert((columns.at<1, 2>() == 10.));
73 assert((columns * diagonal == band));
74
75 // An upper triangular denominator substitutes forward along the row.
76 constexpr sparse::pattern first_of_row{sparse::shape<1, 3>, {{0, 0}}};
77
78 const sparse_matrix<double, first_of_row> first{{2., 0_o, 0_o}};
79 const sparse_matrix<double, sparse::upper_triangular<3>> upper{
80 {2., 1., 0.5}, {0_o, 4., 1.}, {0_o, 0_o, 0.5}};
81 const auto substituted{first / upper};
82
83 assert((substituted.at<0, 0>() == 1.));
84 assert((substituted.at<0, 1>() == -0.25));
85 assert((substituted.at<0, 2>() == -0.5));
86 assert((substituted * upper == first));
87
88 // A lower triangular denominator divides the identity into its inverse.
89 const sparse_matrix<double, sparse::diagonal<2>> identity{1., 1.};
90 const sparse_matrix<double, sparse::lower_triangular<2>> lower{{2., 0_o},
91 {1., 4.}};
92 const auto inverse{identity / lower};
93
94 static_assert(
95 std::same_as<std::remove_cvref_t<decltype(inverse)>,
96 sparse_matrix<double, sparse::lower_triangular<2>>>);
97 assert((inverse.at<0, 0>() == 0.5));
98 assert((inverse.at<1, 0>() == -0.125));
99 assert((inverse.at<1, 1>() == 0.25));
100 assert((inverse * lower == identity));
101
102 // A dense denominator whose elimination exchanges its rows to pivot on the
103 // largest element, of a rectangular numerator.
104 const sparse_matrix<double, sparse::dense<2, 2>> dense{{1., 2.}, {1., 4.}};
105 const sparse_matrix<double, sparse::dense<3, 2>> rectangular{
106 {1., 0.}, {0., 1.}, {1., 1.}};
107 const auto solved{rectangular / dense};
108
109 assert((solved.at<0, 0>() == 2.));
110 assert((solved.at<0, 1>() == -1.));
111 assert((solved.at<1, 0>() == -0.5));
112 assert((solved.at<1, 1>() == 0.5));
113 assert((solved.at<2, 0>() == 1.5));
114 assert((solved.at<2, 1>() == -0.5));
115 assert((solved * dense == rectangular));
116
117 // A block triangular denominator solves its last unknown first, then the
118 // irreducible block of the others, updated by it.
119 constexpr sparse::pattern coupled{
120 sparse::shape<3, 3>, {{0, 0}, {0, 1}, {1, 0}, {1, 1}, {2, 1}, {2, 2}}};
121
122 const sparse_matrix<double, sparse::dense<1, 3>> row{{1., 1., 2.}};
123 const sparse_matrix<double, coupled> blocks{
124 {1., 2., 0_o}, {1., 4., 0_o}, {0_o, 1., 2.}};
125 const auto unknowns{row / blocks};
126
127 assert((unknowns.at<0, 0>() == 2.));
128 assert((unknowns.at<0, 1>() == -1.));
129 assert((unknowns.at<0, 2>() == 1.));
130 assert((unknowns * blocks == row));
131
132 // An irreducible tridiagonal denominator has a dense inverse.
133 const sparse_matrix<double, sparse::diagonal<3>> ones{1., 1., 1.};
134 const tridiagonal_matrix stencil{1., 1., 1., 2., 1., 1., 2.};
135 const auto filled{ones / stencil};
136
137 static_assert(std::same_as<std::remove_cvref_t<decltype(filled)>,
138 sparse_matrix<double, sparse::dense<3, 3>>>);
139 assert((filled.at<0, 0>() == 3.));
140 assert((filled.at<0, 2>() == 1.));
141 assert((filled.at<1, 1>() == 2.));
142 assert((filled.at<2, 1>() == -1.));
143 assert((filled * stencil == ones));
144
145 // A lower bidiagonal denominator of more blocks than are solved of constant
146 // table ranges: a substitution from the last unknown, alternating.
147 constexpr std::size_t size{20};
148 sparse_matrix<double, sparse::banded<size, size, 1, 0>> chain;
149 sparse_matrix<double, sparse::dense<1, size>> sums;
150
151 for (double &element : chain.values) {
152 element = 1.;
153 }
154
155 for (double &element : sums.values) {
156 element = 1.;
157 }
158
159 const auto alternating{sums / chain};
160
161 assert((alternating.at<0, size - 1>() == 1.));
162 assert((alternating.at<0, size - 2>() == 0.));
163 assert((alternating.at<0, 0>() == 0.));
164 assert((alternating * chain == sums));
165
166 return 0;
