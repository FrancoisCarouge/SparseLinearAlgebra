54template <typename Type, auto Lhs, auto Rhs>
55[[nodiscard]] constexpr auto combine(const sparse_matrix<Type, Lhs> &lhs,
56 const sparse_matrix<Type, Rhs> &rhs,
57 const auto &both, const auto &rhs_only)
58 -> sparse_matrix<Type, sparse::pattern_union<Lhs, Rhs>> {
59 constexpr auto combined{sparse::pattern_union<Lhs, Rhs>};
60 sparse_matrix<Type, combined> result;
61
62 for (std::size_t i{0}; i < combined.count; ++i) {
63 const std::size_t lhs_index{projection<Lhs, combined>[i]};
64 const std::size_t rhs_index{projection<Rhs, combined>[i]};
65
66 if (lhs_index == sparse::absent) {
67 result.values[i] = rhs_only(rhs.values[rhs_index]);
68 } else if (rhs_index == sparse::absent) {
69 result.values[i] = lhs.values[lhs_index];
70 } else {
71 result.values[i] = both(lhs.values[lhs_index], rhs.values[rhs_index]);
72 }
73 }
74
75 return result;
76}
